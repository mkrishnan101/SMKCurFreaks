// # Prob 1:
// # Write a program which takes two values x and y. Prints x for y number of times.


#include <iostream>

using namespace std;

int main() {
    // Write C++ code here

    int x,y;


        cout << "Enter an integer: ";
        cin >> x;

        cout << "Enter another integer: ";
        cin >> y;

        for(int i=1; i <= y; i++){
             cout << i << ") " <<x <<"\n" ;
        }
 
       

    return 0;
}
============
Enter an integer: 10
Enter another integer: 5
1) 10
2) 10
3) 10
4) 10
5) 10
============
